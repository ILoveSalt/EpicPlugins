
DROP DATABASE IF EXISTS hotel_comfort;
CREATE DATABASE hotel_comfort CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE hotel_comfort;

CREATE TABLE room_categories (
    category_id INT AUTO_INCREMENT PRIMARY KEY,
    category_name VARCHAR(50) NOT NULL UNIQUE,
    base_price_per_night DECIMAL(10, 2) NOT NULL CHECK (base_price_per_night > 0),
    description VARCHAR(255)
);

CREATE TABLE rooms (
    room_id INT AUTO_INCREMENT PRIMARY KEY,
    room_number VARCHAR(10) NOT NULL UNIQUE,
    category_id INT NOT NULL,
    floor INT NOT NULL CHECK (floor > 0),
    capacity INT NOT NULL CHECK (capacity > 0),
    status ENUM('свободен', 'занят', 'уборка') NOT NULL DEFAULT 'свободен',
    FOREIGN KEY (category_id) REFERENCES room_categories(category_id)
        ON UPDATE CASCADE ON DELETE RESTRICT
);

CREATE TABLE guests (
    guest_id INT AUTO_INCREMENT PRIMARY KEY,
    full_name VARCHAR(150) NOT NULL,
    passport_data VARCHAR(50) NOT NULL UNIQUE,
    phone VARCHAR(20) NOT NULL UNIQUE,
    email VARCHAR(100),
    citizenship VARCHAR(60) NOT NULL DEFAULT 'РФ'
);

CREATE TABLE bookings (
    booking_id INT AUTO_INCREMENT PRIMARY KEY,
    guest_id INT NOT NULL,
    room_id INT NOT NULL,
    check_in_date DATE NOT NULL,
    check_out_date DATE NOT NULL,
    guests_count INT NOT NULL CHECK (guests_count > 0),
    total_cost DECIMAL(10, 2) NOT NULL DEFAULT 0.00 CHECK (total_cost >= 0),
    status ENUM('забронирован', 'заселён', 'выселен') NOT NULL DEFAULT 'забронирован',
    FOREIGN KEY (guest_id) REFERENCES guests(guest_id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    FOREIGN KEY (room_id) REFERENCES rooms(room_id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT chk_dates CHECK (check_out_date > check_in_date)
);

CREATE TABLE services (
    service_id INT AUTO_INCREMENT PRIMARY KEY,
    service_name VARCHAR(100) NOT NULL UNIQUE,
    price DECIMAL(10, 2) NOT NULL CHECK (price >= 0)
);

CREATE TABLE booking_services (
    booking_service_id INT AUTO_INCREMENT PRIMARY KEY,
    booking_id INT NOT NULL,
    service_id INT NOT NULL,
    quantity INT NOT NULL DEFAULT 1 CHECK (quantity > 0),
    FOREIGN KEY (booking_id) REFERENCES bookings(booking_id)
        ON UPDATE CASCADE ON DELETE CASCADE,
    FOREIGN KEY (service_id) REFERENCES services(service_id)
        ON UPDATE CASCADE ON DELETE RESTRICT
);
